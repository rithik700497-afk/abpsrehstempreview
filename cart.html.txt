<script>
let cart = JSON.parse(localStorage.getItem("cart")) || {};

function render(){
 let html="", subtotal=0;

 for(let i in cart){
  let item=cart[i];
  subtotal += item.price * item.qty;

  html += `
  <div class="item">
    <img src="${item.img}">
    <div class="details">
      <b>${i}</b><br>
      <span class="price">₹${item.price}</span>
    </div>
    <div class="qty">
      <button onclick="change('${i}',-1)">-</button>
      ${item.qty}
      <button onclick="change('${i}',1)">+</button>
    </div>
  </div>`;
 }

 document.getElementById("items").innerHTML = html;
 document.getElementById("sub").innerText = "₹"+subtotal;
 document.getElementById("total").innerText = "₹"+(subtotal+20);
}

function change(name,val){
 cart[name].qty += val;
 if(cart[name].qty <= 0) delete cart[name];

 localStorage.setItem("cart", JSON.stringify(cart));
 render();
}

render();
</script>
